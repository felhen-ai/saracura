def save():
 p=STATE.with_suffix('.tmp');p.write_text(json.dumps({'updated':datetime.datetime.now(datetime.timezone.utc).isoformat(),'models':records},indent=2));p.replace(STATE)
def gethealth():
 return json.load(urllib.request.urlopen('http://127.0.0.1:18031/health',timeout=4))
def unit(a):return 'saracura-round-'+a['id'].split('/')[-1].lower().replace('.','-')
def stop(a):subprocess.run(['systemctl','--user','stop',unit(a)],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL)
def cleanup(a):
 p=R/'hub-queue'
 # This cache is exclusively owned by this sequential queue. Modern HF puts blobs at its root.
 if p.exists():
  assert p.parent==R and not p.is_symlink();shutil.rmtree(p)
def download(a):
 # Explicit authorized download in a subprocess with networking enabled only for this step.
 cache=R/'hub-queue'
 allocated=sum(p.stat().st_blocks*512 for p in cache.rglob('*') if p.is_file() and not p.is_symlink()) if cache.exists() else 0
 if shutil.disk_usage(R).free+allocated < (a['bf16_weights_gib_estimate']+10)*1024**3:raise RuntimeError('insufficient_disk_reserve')
 code='from huggingface_hub import snapshot_download;import sys;snapshot_download(sys.argv[1],revision=sys.argv[2],cache_dir=sys.argv[3],allow_patterns=["*.safetensors","*.json","*.jinja","*.model","tokenizer*"],token=False)'
 env=dict(os.environ);env.pop('HF_HUB_OFFLINE',None);env.pop('TRANSFORMERS_OFFLINE',None)
 with (R/(a['id'].replace('/','--')+'-download.log')).open('w') as log:
  subprocess.run([PY,'-c',code,a['id'],a['revision'],str(R/'hub-queue')],env=env,stdout=log,stderr=log,check=True)
def preflight(a):
 import transformers
 from transformers import AutoTokenizer
 tokenizer=AutoTokenizer.from_pretrained(a['id'],revision=a['revision'],local_files_only=True,trust_remote_code=False)
 config=bridge.Config(model=a['id'],revision=a['revision'],device='cpu',context=8192,threads=2,template_kwargs=KW.get(a['id'],{}),dtype='bfloat16')
 kwargs=dict(config.template_kwargs)
 if 'date_string' in tokenizer.chat_template:kwargs['date_string']='2026-10-21'
 clock=__import__('transformers.utils.chat_template_utils',fromlist=['x']) if 'strftime_now' in tokenizer.chat_template else None
 backend=bridge.TransformersBackend(None,transformers,tokenizer,None,config,kwargs,clock);backend.probe_clock();backend.prepare_letter_tokens()
 rng=random.Random(bench.SEED);maximum=0;count=0;inputs=[]
 for task in bench.TASKS:
  raw,rows=bench.fetch(M['dataset']['revision'],task,pathlib.Path.home()/'.cache/ptbr-typed-decisions-bench')
  for row in rows:
   q=bench.question(row,rng);question=bridge.parse_question('q',q);text=bridge.render_user_message(row['text'][:3500],question);n=len(backend.encode(text));maximum=max(maximum,n);count+=1
 if maximum+1>8192:raise RuntimeError('context_overflow_preflight')
 result={'items':count,'max_prompt_tokens':maximum,'context':8192,'additional_truncation':False,'template_kwargs':kwargs,'status':'pass'}
 (R/(a['id'].replace('/','--')+'-preflight.json')).write_text(json.dumps(result,indent=2));return result
def start(a,label):
 name=unit(a);active=subprocess.run(['systemctl','--user','is-active',name],capture_output=True,text=True)
 if active.stdout.strip()=='active':raise RuntimeError('own_unit_already_active')
 ev=R/(a['id'].replace('/','--')+'-'+label+'-constrained-letter-evidence.json')
 if ev.exists():raise RuntimeError('existing_evidence_refuse_overwrite')
 args=['systemd-run','--user','--unit='+name,'--property=MemoryMax=20G','--property=MemorySwapMax=0','--property=CPUQuota=250%','--setenv=CUDA_VISIBLE_DEVICES=','--setenv=HF_HUB_CACHE='+str(R/'hub-queue'),'--setenv=OMP_NUM_THREADS=2',PY,'-u',str(R/'llm_bridge_constrained.py'),'--model',a['id'],'--revision',a['revision'],'--device','cpu','--dtype','bfloat16','--threads','2','--port','18031','--context','8192','--readout','constrained-letter','--generation-timeout','180','--chat-template-kwargs',json.dumps(KW.get(a['id'],{})),'--evidence-output',str(ev)]
 if a['id']=='Qwen/Qwen3.5-4B':args+=['--auto-class','AutoModelForImageTextToText']
 subprocess.run(args,check=True,capture_output=True)
 until=time.monotonic()+240
 while time.monotonic()<until:
  try:
   h=gethealth()
   if h['evidence']['model']['revision']!=a['revision']:raise RuntimeError('wrong_server_identity')
   return h
  except (OSError,ValueError):pass
  st=subprocess.run(['systemctl','--user','is-active',name],capture_output=True,text=True).stdout.strip()
  if st=='failed':raise RuntimeError('server_start_failed')
  time.sleep(2)
 raise RuntimeError('startup_timeout')
def evaluate(a,label,h,limit=None):
 slug=a['id'].replace('/','--');output=R/(slug+'-'+label+'-constrained-letter.json')
 if output.exists():raise RuntimeError('existing_report_refuse_overwrite')
 args=[PY,'-u',str(R/'ptbr_bench.py'),'--base-url','http://127.0.0.1:18031','--model',a['id'],'--name',slug.lower()+'-'+label+'-constrained-letter','--output',str(output),'--revision',M['dataset']['revision'],'--checkpoint-id',a['id'],'--checkpoint-revision',a['revision'],'--subfolder','.','--context','8192 context max_output1 no_thinking constrained-letter','--hardware','Ryzen7 7700X CPU2threads capped250percent','--quantization','none bf16','--server','ptbr-llm-bridge-constrained-letter','--server-version',h['evidence_sha256'],'--timeout','240']
 if limit:args+=['--limit',str(limit)]
 with (R/(slug+'-'+label+'-constrained-letter.log')).open('x') as log:
  client=subprocess.Popen(args,stdout=log,stderr=log)
  try:
   while client.poll() is None:
    active=subprocess.run(['systemctl','--user','is-active',unit(a)],capture_output=True,text=True).stdout.strip()
    if active!='active':raise RuntimeError('model_unit_exited_during_evaluation')
    time.sleep(5)
   if client.returncode:raise RuntimeError('evaluator_exit_'+str(client.returncode))
  finally:
   if client.poll() is None:
    client.terminate()
    try:client.wait(timeout=10)
    except subprocess.TimeoutExpired:client.kill();client.wait()

 a2=json.load(open(output));after=gethealth();assert after['evidence_sha256']==h['evidence_sha256'];assert h['evidence']['readout']=='constrained-letter';assert a2['checkpoint']['server']=='ptbr-llm-bridge-'+h['evidence']['readout']
 failed=sum(t['failed'] for t in a2['tasks'].values());result={'state':'pass' if not failed else 'failed','items':sum(t['n'] for t in a2['tasks'].values()),'failed':failed,'report_sha256':hashlib.sha256(output.read_bytes()).hexdigest(),'evidence_sha256':h['evidence_sha256'],'counters':after['counters']}
 (R/(slug+'-'+label+'-constrained-letter-sidecar.json')).write_text(json.dumps(result,indent=2));return result
passing=[]
models=[x for x in M['models'] if x['cohort']=='generated_decision']
