 from contextlib import asynccontextmanager

   sys.path.insert(0,str(R))
   import ptbr_bench as b
   manifest=json.load(open(R/'manifest.json'));rng=random.Random(b.SEED);count=0;maximum=0;state_max=0

    assert len(states)==1;engine=states[0]['engine']
   for task in b.TASKS:
    _,rows=b.fetch(manifest['dataset']['revision'],task,pathlib.Path.home()/'.cache/ptbr-typed-decisions-bench')

     q=b.question(row,rng);state=row['text'][:3500]

     state_max=max(state_max,n);maximum=max(maximum,width);count+=1
   result={'status':'pass','items':count,'max_record_tokens':maximum,'max_state_tokens':state_max,'state_truncated':False,'upstream_preprocessing':'candidate readout clips candidate tokens to64 in lev' if kind=='lev' else 'original decider schema/state preparation; state cap8192 not reached','base_revision':base_revision}
   dest=R/(pathlib.Path(path).name+'-cpu-reference-preflight.json');payload=json.dumps(result,indent=2)

    with dest.open('x') as f:f.write(payload)
   print('preflight '+str(result),flush=True)
